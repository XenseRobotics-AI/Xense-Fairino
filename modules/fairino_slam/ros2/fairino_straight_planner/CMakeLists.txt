cmake_minimum_required(VERSION 3.16)
project(fairino_straight_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ament_cmake REQUIRED)
set(dependencies nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle
  pluginlib geometry_msgs nav_msgs tf2_ros)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/planner.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${dependencies})
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
add_executable(plugin_check src/plugin_check.cpp)
ament_target_dependencies(plugin_check nav2_core pluginlib)
install(TARGETS plugin_check DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_core test/test_core.cpp)
  target_include_directories(test_core PRIVATE include)
  target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME core COMMAND test_core)
  add_executable(test_planner_api test/test_planner_api.cpp)
  target_include_directories(test_planner_api PRIVATE include)
  target_compile_options(test_planner_api PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME planner_api COMMAND test_planner_api)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_plugin test/test_plugin.cpp)
  if(TARGET test_plugin)
    target_link_libraries(test_plugin ${PROJECT_NAME})
    ament_target_dependencies(test_plugin ${dependencies})
  endif()
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies})
ament_package()
